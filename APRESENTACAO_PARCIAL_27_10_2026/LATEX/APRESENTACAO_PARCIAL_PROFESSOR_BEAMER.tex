\begin{frame}{Problema proposto}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.58\textwidth}
      \large Como antecipar a quantidade semanal de internações por VSR antes da chegada de uma onda sazonal?
      \vspace{.45cm}
      \begin{itemize}
        \item O foco é previsão semanal, não explicação causal de todos os determinantes da doença.
        \item O resultado deve apoiar planejamento de leitos pediátricos, equipes e insumos.
        \item A avaliação precisa reproduzir as informações que estariam disponíveis na data de cada previsão.
      \end{itemize}
    \end{column}
    \begin{column}{.38\textwidth}
      \begin{block}{Unidade de análise}
        Número semanal de internações confirmadas por VSR, observado por UF, região e Brasil.
      \end{block}
      \vspace{.2cm}
      \textcolor{pucblue}{\large\textbf{Pergunta central}}\\[.1cm]
      Previsões de 1, 4, 8 e 13 semanas podem antecipar a demanda melhor do que referências simples?
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Referências e modelos candidatos}
  \begin{block}{Referência sazonal obrigatória}
    \[
      \widehat{y}_{t+h} = y_{t+h-52}
    \]
    A previsão usa a semana epidemiológica equivalente da temporada anterior. Um modelo complexo precisa superar essa referência de modo consistente.
  \end{block}
  \vspace{.35cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.23\textwidth}\textcolor{pucblue}{\textbf{Ingênuo}}\\ \small Mantém o último valor observado.\end{column}
    \begin{column}{.23\textwidth}\textcolor{pucblue}{\textbf{ETS / Holt-Winters}}\\ \small Representa nível, tendência e sazonalidade.\end{column}
    \begin{column}{.23\textwidth}\textcolor{pucblue}{\textbf{SARIMA}}\\ \small Representa dependência temporal e sazonal.\end{column}
    \begin{column}{.23\textwidth}\textcolor{pucblue}{\textbf{Hierárquico}}\\ \small Busca coerência entre UF, região e Brasil.\end{column}
  \end{columns}
  \vspace{.5cm}
  \small Regressão dinâmica será considerada somente se covariáveis externas tiverem qualidade e estiverem disponíveis na data da previsão.
\end{frame}

\begin{frame}{Validação temporal}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.56\textwidth}
      \includegraphics[width=\textwidth,height=.69\textheight,keepaspectratio]{calendario_validacao_brasil.png}
    \end{column}
    \begin{column}{.40\textwidth}
      \begin{itemize}
        \item 58 cortes por série.
        \item Treino inicial de 156 semanas, aproximadamente três anos.
        \item Janela de treino expansiva.
        \item Horizontes de 1, 4, 8 e 13 semanas.
      \end{itemize}
      \vspace{.25cm}
      \small Cada corte simula uma data histórica. O modelo vê somente o passado e depois é comparado ao valor realmente observado.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Métricas de avaliação}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.31\textwidth}
      \textcolor{pucblue}{\Large\textbf{MAE}}\\[.1cm]
      \small Erro absoluto médio. Informa, em média, quantos casos a previsão erra.
    \end{column}
    \begin{column}{.31\textwidth}
      \textcolor{pucblue}{\Large\textbf{RMSE}}\\[.1cm]
      \small Dá peso maior a erros grandes, especialmente relevantes durante picos.
    \end{column}
    \begin{column}{.31\textwidth}
      \textcolor{pucblue}{\Large\textbf{MASE}}\\[.1cm]
      \small Escala o erro por uma referência e permite comparar séries de magnitudes distintas.
    \end{column}
  \end{columns}
  \vspace{.8cm}
  \begin{block}{Decisão metodológica}
    As métricas serão calculadas separadamente por horizonte. MAPE não será métrica principal porque várias UFs possuem semanas com zero caso.
  \end{block}
\end{frame}
