\begin{frame}{Problema proposto}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.57\textwidth}
      \begin{mensagem}
        \large Como antecipar a quantidade semanal de internações por VSR
        antes da chegada de uma onda sazonal?
      \end{mensagem}
      \vspace{.2cm}
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item O foco é previsão semanal, não explicação causal de todos os determinantes da doença.
        \item O resultado deve apoiar planejamento de leitos pediátricos, equipes e insumos.
        \item A avaliação precisa reproduzir as informações que estariam disponíveis na data de cada previsão.
      \end{itemize}
    \end{column}
    \begin{column}{.38\textwidth}
      \begin{cartao}
        \rotulo{Unidade de análise}
        \small Número semanal de internações confirmadas por VSR, observado por UF, região e Brasil.
      \end{cartao}
      \vspace{.15cm}
      \begin{cartao}[borderline west={3pt}{0pt}{pucorange}]
        \rotulo{Pergunta central}
        \small Previsões de 1, 4, 8 e 13 semanas podem antecipar a demanda melhor do que referências simples?
      \end{cartao}
    \end{column}
  \end{columns}

\begin{frame}{Referências e modelos candidatos}

\begin{frame}{Métricas de avaliação}
  \vspace{.2cm}
  \begin{columns}[T,onlytextwidth]

      \begin{cartao}[height=2.6cm]
        {\color{pucblue}\LARGE\bfseries MAE}\par\vspace{4pt}
        \small Erro absoluto médio. Informa, em média, quantos casos a previsão erra.
      \end{cartao}
    \end{column}
    \begin{column}{.315\textwidth}
      \begin{cartao}[height=2.6cm]
        {\color{pucblue}\LARGE\bfseries RMSE}\par\vspace{4pt}
        \small Dá peso maior a erros grandes, especialmente relevantes durante picos.
      \end{cartao}
    \end{column}
    \begin{column}{.315\textwidth}
      \begin{cartao}[height=2.6cm]
        {\color{pucblue}\LARGE\bfseries MASE}\par\vspace{4pt}
        \small Escala o erro por uma referência e permite comparar séries de magnitudes distintas.
      \end{cartao}
    \end{column}
  \end{columns}
  \vspace{.5cm}
  \begin{mensagem}
    {\color{pucorange}\small\bfseries DECISÃO METODOLÓGICA}\par\vspace{3pt}
    As métricas serão calculadas separadamente por horizonte. MAPE não será métrica
    principal porque várias UFs possuem semanas com zero caso.
  \end{mensagem}
\end{frame}
